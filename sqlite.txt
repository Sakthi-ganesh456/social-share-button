package com.example.sqlite_connection;
import android.content.ContentValues; import android.content.Context; import android.database.Cursor;
import android.database.sqlite.SQLiteDatabase; import android.database.sqlite.SQLiteOpenHelper;
public class StudentDatabaseHelper extends SQLiteOpenHelper {
public StudentDatabaseHelper(Context context) { super(context, "Student.db", null, 1);
}
@Override
public void onCreate(SQLiteDatabase db) {
db.execSQL("CREATE TABLE Student(" +
"id INTEGER PRIMARY KEY AUTOINCREMENT," + "name TEXT," + "age INTEGER," +
"department TEXT" + ");");
}
@Override
public void onUpgrade(SQLiteDatabase db, int oldVersion, int newVersion) { db.execSQL("DROP TABLE IF EXISTS Student");
onCreate(db);
}
// INSERT
public void insertStudent(String name, int age, String department) { SQLiteDatabase db = this.getWritableDatabase();
ContentValues values = new ContentValues();
values.put("name", name); values.put("age", age); values.put("department", department); db.insert("Student", null, values);
db.close();
}

// READ
public Cursor getAllStudents() {
SQLiteDatabase db = this.getReadableDatabase();
Cursor cursor = db.rawQuery( "SELECT * FROM Student", null
);
return cursor;
}
// UPDATE
public void updateStudent( int id, String name, int age, String department) {
SQLiteDatabase db = this.getWritableDatabase();
ContentValues values = new ContentValues();
values.put("name", name); values.put("age", age); values.put("department", department);
db.update(
"Student", values, "id = ?",
new String[]{String.valueOf(id)}
);
db.close();
}
// DELETE
public void deleteStudent(int id) {
SQLiteDatabase db = this.getWritableDatabase(); db.delete(
"Student", "id = ?",
new String[]{String.valueOf(id)}
);
db.close();
}
}
